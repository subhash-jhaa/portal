type Screen = 'login' | 'dashboard' | 'profile' | 'payslip'
type DisplayState = 'normal' | 'loading' | 'empty' | 'error'

const nav = [
  { id: 'dashboard' as Screen, label: 'Overview', icon: Home },
  { id: 'profile' as Screen, label: 'My profile', icon: UserRound },
  { id: 'payslip' as Screen, label: 'Payslips', icon: ReceiptIndianRupee },
]

export function GrowmerPortal() {
  const [screen, setScreen] = useState<Screen>('dashboard')
  const [option, setOption] = useState<1 | 2>(1)
  const [state, setState] = useState<DisplayState>('normal')
  const [dark, setDark] = useState(false)
  const [mobileNav, setMobileNav] = useState(false)

  const selectScreen = (next: Screen) => { setScreen(next); setOption(1); setState('normal'); setMobileNav(false) }

  return (
    <div className={cn('min-h-screen bg-background text-foreground transition-colors', dark && 'dark')}>
      <div className="flex min-h-screen">
        {screen !== 'login' && <Sidebar screen={screen} selectScreen={selectScreen} mobileNav={mobileNav} setMobileNav={setMobileNav} />}
        <main className={cn('min-w-0 flex-1', screen === 'login' ? 'flex flex-col' : 'lg:pl-0')}>
          {screen !== 'login' && <Topbar dark={dark} setDark={setDark} setMobileNav={setMobileNav} />}
          {screen === 'login' ? <Login option={option} setOption={setOption} state={state} setState={setState} dark={dark} setDark={setDark} /> : (
            <div className="mx-auto w-full max-w-[1440px] p-4 sm:p-6 lg:p-10">
              <PreviewBar option={option} setOption={setOption} state={state} setState={setState} />
              {state === 'loading' ? <LoadingState /> : state === 'empty' ? <EmptyState screen={screen} /> : state === 'error' ? <ErrorState setState={setState} /> : (
                screen === 'dashboard' ? <Dashboard option={option} selectScreen={selectScreen} /> : screen === 'profile' ? <Profile option={option} /> : <Payslip option={option} />
              )}
            </div>
          )}
        </main>
      </div>
    </div>
  )
}
